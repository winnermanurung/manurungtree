function doGet() {

function initSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetAnggota = ss.getSheetByName('Anggota') || ss.insertSheet('Anggota');
  if (sheetAnggota.getLastRow() === 0) {
    sheetAnggota.appendRow(['id', 'nama', 'generasi', 'status', 'parentId', 'hp', 'kota', 'foto']);
  }
  var sheetKas = ss.getSheetByName('Kas') || ss.insertSheet('Kas');
  if (sheetKas.getLastRow() === 0) {
    sheetKas.appendRow(['id', 'tipe', 'memberId', 'bulan', 'tahun', 'nominal', 'keterangan', 'tanggal']);
  }
}

function getInitialData() {
  initSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return {
    anggota: getSheetData(ss.getSheetByName('Anggota')),
    kas: getSheetData(ss.getSheetByName('Kas'))
  };
}

function getSheetData(sheet) {
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    result.push(obj);
  }
  return result;
}

function saveAnggotaData(data) {
  initSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Anggota');
  var rows = sheet.getDataRange().getValues();
  var foundRow = -1;
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] == data.id) { foundRow = i + 1; break; }
  }
  var rowData = [data.id, data.nama, data.generasi, data.status, data.parentId, data.hp, data.kota, data.foto];
  if (foundRow > 0) {
    sheet.getRange(foundRow, 1, 1, 8).setValues([rowData]);
  } else {
    sheet.appendRow(rowData);
  }
  return { status: "success" };
}

function deleteAnggotaData(id) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Anggota');
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] == id) { sheet.deleteRow(i + 1); break; }
  }
  return { status: "success" };
}

function saveKasData(data) {
  initSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Kas');
  sheet.appendRow([data.id, data.tipe, data.memberId, data.bulan, data.tahun, data.nominal, data.keterangan, data.tanggal]);
  return { status: "success" };
}

function deleteKasData(id) {

}
