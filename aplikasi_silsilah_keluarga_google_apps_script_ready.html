Panduan Deploy ke Google Apps Script

Ikuti langkah mudah di bawah untuk menyambungkan aplikasi ini dengan Google Sheets sebagai Database permanen gratis.

Langkah Pemasangan di Google Drive:

  1. Buat Google Sheet baru di Google Drive (misal beri nama: DB_Silsilah_Keluarga).
  2. Buka menu Ekstensi > Apps Script.
  3. Salin kode Code.gs di bawah ke file Code.gs di Apps Script editor.
  4. Buat file HTML baru bernama Index.html di Apps Script editor dan salin seluruh isi file HTML ini.
  5. Klik tombol Terapkan (Deploy) > Web app baru (New deployment). Setel akses ke "Anyone" (Siapa Saja).

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
      .setTitle('Traharjo - Pohon Silsilah Keluarga')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

// Mengambil seluruh data dari Google Spreadsheet
function getFamilyData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Members');
  if (!sheet) return [];
  
  var data = sheet.getDataRange().getValues();
  var headers = data.shift();
  
  return data.map(function(row) {
    var obj = {};
    headers.forEach(function(header, i) {
      obj[header] = row[i];
    });
    return obj;
  });
}

// Menyimpan atau memperbarui data anggota
function saveMemberData(memberObj) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Members');
  if (!sheet) {
    sheet = ss.insertSheet('Members');
    sheet.appendRow(['id', 'fullName', 'nickname', 'gender', 'birthDate', 'deathDate', 'isLiving', 'generation', 'fatherId', 'motherId', 'spouseId', 'photoUrl', 'phone', 'address', 'notes']);
  }
  
  var data = sheet.getDataRange().getValues();
  var rowIndex = -1;
  
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] == memberObj.id) {
      rowIndex = i + 1;
      break;
    }
  }
  
  var row = [
    memberObj.id,
    memberObj.fullName,
    memberObj.nickname,
    memberObj.gender,
    memberObj.birthDate,
    memberObj.deathDate || '',
    memberObj.isLiving ? 'TRUE' : 'FALSE',
    memberObj.generation,
    memberObj.fatherId || '',
    memberObj.motherId || '',
    memberObj.spouseId || '',
    memberObj.photoUrl || '',
    memberObj.phone || '',
    memberObj.address || '',
    memberObj.notes || ''
  ];
  
  if (rowIndex > -1) {
    sheet.getRange(rowIndex, 1, 1, row.length).setValues([row]);
  } else {
    sheet.appendRow(row);
  }
  
  return { success: true };
}
